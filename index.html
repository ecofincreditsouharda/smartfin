Transaction History

Setup required: The share_txns table has not been created yet.
Run this SQL in Supabase SQL Editor: 
CREATE TABLE IF NOT EXISTS public.share_txns (
  txn_no           TEXT PRIMARY KEY,
  member_id        TEXT NOT NULL,
  member_name      TEXT DEFAULT '',
  branch           TEXT DEFAULT '',
  shares           INTEGER NOT NULL DEFAULT 1,
  amount_per_share NUMERIC(12,2) NOT NULL DEFAULT 0,
  total_amount     NUMERIC(12,2) NOT NULL DEFAULT 0,
  date             DATE NOT NULL DEFAULT CURRENT_DATE,
  note             TEXT DEFAULT '',
  entered_by       TEXT DEFAULT '',
  created_at       TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE public.share_txns ENABLE ROW LEVEL SECURITY;
CREATE POLICY "service_role_all" ON public.share_txns
  FOR ALL TO service_role USING (true) WITH CHECK (true);
GRANT ALL ON public.share_txns TO service_role;
